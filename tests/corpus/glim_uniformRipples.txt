// "u_" uniform style: u_time, u_resolution and u_mouse (in pixels), with a loop
// that writes one colour channel per pass.
#ifdef GL_ES
precision mediump float;
#endif

uniform float u_time;
uniform vec2 u_resolution;
uniform vec2 u_mouse;

void main() {
    float aspect = u_resolution.x / u_resolution.y;
    vec2 st = gl_FragCoord.xy / u_resolution.xy;
    st.x *= aspect;
    vec2 m = u_mouse / u_resolution;
    m.x *= aspect;
    vec3 color = vec3(0.0);
    for (int i = 0; i < 3; i++) {
        float a = u_time * 0.6 + float(i) * 2.094;
        vec2 c = m + 0.25 * vec2(cos(a), sin(a));
        color[i] = 0.5 + 0.5 * sin(distance(st, c) * 40.0 - u_time * 4.0);
    }
    gl_FragColor = vec4(color * color, 1.0);
}
