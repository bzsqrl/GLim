const float speed = 1.0;

void mainImage( out vec4 fragColor, in vec2 fragCoord ) {
    vec2 p = (2.0 * fragCoord.xy - iResolution.xy) / iResolution.y;
    float t = iTime * speed;
    
    p *= 3.0;
    float v = sin(p.x + sin(p.y + t));
    v += sin(p.y + sin(p.x + t * 0.8));
    v += sin(p.x + p.y + t * 0.5);
    
    vec3 col = vec3(1.0, 0.6, 0.7); // Pinkish
    col *= 0.5 + 0.5 * v;
    
    // Shadows
    col *= smoothstep(-1.0, 1.0, v);
    
    fragColor = vec4(col, 1.0);
}
