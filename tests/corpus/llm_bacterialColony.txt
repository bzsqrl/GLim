const float speed = 1.0;

void mainImage( out vec4 fragColor, in vec2 fragCoord ) {
    vec2 p = (2.0 * fragCoord.xy - iResolution.xy) / iResolution.y;
    float t = iTime * speed;
    
    float d = 0.0;
    for(int i = 0; i < 20; i++) {
        float fi = float(i);
        vec2 pos = vec2(
            sin(fi * 132.0 + t * 0.1),
            cos(fi * 453.0 + t * 0.1)
        ) * (0.5 + 0.4 * sin(t * 0.2 + fi));
        d += 0.02 / length(p - pos);
    }
    
    vec3 col = vec3(0.2, 0.8, 0.4) * smoothstep(0.8, 1.0, d);
    col += vec3(0.1, 0.3, 0.2) * smoothstep(0.2, 0.8, d);
    
    fragColor = vec4(col, 1.0);
}
