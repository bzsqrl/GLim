#define TIME        iTime
#define RESOLUTION  iResolution
#define PI          3.141592654
#define TAU         (2.0*PI)

// License: WTFPL, author: sam hocevar, found: https://stackoverflow.com/a/17897228/418488
const vec4 hsv2rgb_K = vec4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);
vec3 hsv2rgb(vec3 c) {
  vec3 p = abs(fract(c.xxx + hsv2rgb_K.xyz) * 6.0 - hsv2rgb_K.www);
  return c.z * mix(hsv2rgb_K.xxx, clamp(p - hsv2rgb_K.xxx, 0.0, 1.0), c.y);
}
// License: WTFPL, author: sam hocevar, found: https://stackoverflow.com/a/17897228/418488
//  Macro version of above to enable compile-time constants
#define HSV2RGB(c)  (c.z * mix(hsv2rgb_K.xxx, clamp(abs(fract(c.xxx + hsv2rgb_K.xyz) * 6.0 - hsv2rgb_K.www) - hsv2rgb_K.xxx, 0.0, 1.0), c.y))


// License: MIT, author: Inigo Quilez, found: https://iquilezles.org/articles/noacos/
mat3 rot(vec3 d, vec3 z) {
  vec3  v = cross( z, d );
  float c = dot( z, d );
  float k = 1.0/(1.0+c);

  return mat3( v.x*v.x*k + c,     v.y*v.x*k - v.z,    v.z*v.x*k + v.y,
               v.x*v.y*k + v.z,   v.y*v.y*k + c,      v.z*v.y*k - v.x,
               v.x*v.z*k - v.y,   v.y*v.z*k + v.x,    v.z*v.z*k + c    );
}

// License: MIT, author: Inigo Quilez, found: https://iquilezles.org/articles/intersectors/
vec2 box(vec3 ro, vec3 rd, vec3 boxSize, out vec3 outNormal) {
    vec3 m = 1.0/rd; // can precompute if traversing a set of aligned boxes
    vec3 n = m*ro;   // can precompute if traversing a set of aligned boxes
    vec3 k = abs(m)*boxSize;
    vec3 t1 = -n - k;
    vec3 t2 = -n + k;
    float tN = max( max( t1.x, t1.y ), t1.z );
    float tF = min( min( t2.x, t2.y ), t2.z );
    if( tN>tF || tF<0.0) return vec2(-1.0); // no intersection
    outNormal = (tN>0.0) ? step(vec3(tN),t1) : // ro ouside the box
                           step(t2,vec3(tF)) ;  // ro inside the box
    outNormal *= -sign(rd);
    return vec2( tN, tF );
}

// License: MIT OR CC-BY-NC-4.0, author: mercury, found: https://mercury.sexy/hg_sdf/
vec3 mod3(inout vec3 p, vec3 size) {
  vec3 c = floor((p + size*0.5)/size);
  p = mod(p + size*0.5,size) - size*0.5;
  return c;
}


float steps(float a, float b) {
  float x = a/b;
  float y = floor(x);
  float z = fract(x);
  return (y+z*z)*b;
}

vec3 effect(vec2 p) {
  const vec3 upDir = vec3(0., 1., 0.);
  vec3 ro = vec3(1.);
  float a0 = 0.3*steps(TIME, 1E-1)+123.4;
  float a1 = 0.4*steps(TIME, 1E-1)+12.34;
  vec3 r0 = normalize(vec3(1., sin(sqrt(3.)*a0), sin(a0)));
  vec3 r1 = normalize(vec3(sin(a1), sin(sqrt(2.)*a1), 1.));
  ro *= rot(r0, r1);
  vec3 la = vec3(0.);
  vec3 ww = normalize(la-ro);
  vec3 uu = normalize(cross(upDir, ww));
  vec3 vv = cross(ww, uu);
  vec3 rd = normalize(p.x*uu + p.y*vv + 2.*ww);
  
  vec3 bn;
  vec2 bi = box(ro, rd, vec3(5.), bn); 
  vec3 abn = abs(bn);
  vec3 col = vec3(0.);
  if (bi.y > bi.x) {
    vec3 p3 = ro+rd*bi.y;
    const float lum = 5E-5;
    const float mn = 1E-4;
    const vec3 col0 = HSV2RGB(vec3(0.6, 0.85, lum));
    const vec3 col1 = HSV2RGB(vec3(0.75 , 0.85, lum));
    const vec3 col2 = HSV2RGB(vec3(0.65, 0.85, lum));
    vec3 cp3 = p3;
    vec3 np3 = mod3(cp3, vec3(0.25));
    float a = steps(TIME, 1E-2);
    cp3 += 0.1*sin(p3.yzx+a);
    col += mix(0.1, 1.0, smoothstep(-0.7, 0.5, sin(0.123*np3.x+1.1*a)))*col0/max((cp3.x*cp3.x), mn)*step(abn.x, 0.)*step(abn.y, 0.);
    col += mix(0.2, 1.0, smoothstep(-0.6, 0.5, sin(0.242*np3.y+0.7*a)))*col1/max((cp3.y*cp3.y), mn)*step(abn.y, 0.)*step(abn.z, 0.);
    col += mix(0.1, 1.0, smoothstep(-0.9, 0.5, sin(0.07*np3.z+1.5*a)))*col2/max((cp3.z*cp3.z), mn)*step(abn.z, 0.)*step(abn.x, 0.);
  }
  col = sqrt(col);
  return col;
}

void mainImage( out vec4 fragColor, in vec2 fragCoord ) {
    vec2 p = -1.+2.*fragCoord/RESOLUTION.xy;

    p.x *= RESOLUTION.x/RESOLUTION.y;
    vec3 col = effect(p);

    fragColor = vec4(col,1.0);
}
